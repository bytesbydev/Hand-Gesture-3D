━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Install dependencies:
   $ npm install

2. Run development server:
   $ npm run dev

3. Open in browser:
   → Usually http://localhost:5173

4. Allow camera access when prompted

5. Start using gestures!

See QUICK_START.md for detailed usage instructions.
